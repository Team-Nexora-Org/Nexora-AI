   300→export async function seedDatabase() {
   301→  // Wipe everything (order matters for FKs; cascade handles most)
   302→  await db.auditLog.deleteMany()
   303→  await db.plannerDecision.deleteMany()
   304→  await db.activityMatch.deleteMany()
   305→  await db.executionEvent.deleteMany()
   306→  await db.fieldReport.deleteMany()
   307→  await db.scheduleActivity.deleteMany()
   308→  await db.supervisor.deleteMany()
   309→  await db.project.deleteMany()
   310→  await db.modelEvaluation.deleteMany()
   311→
   312→  const project = await db.project.create({
   313→    data: {
   314→      code: 'UNIT4-EXP',
   315→      name: 'Unit-4 Expansion Project',
   316→      description:
   317→        'Representative Primavera/MS Project export. L5/L6 planned activities across Civil, Mechanical, Piping, Electrical and Instrumentation disciplines.',
   318→      scheduleSource: 'Representative Primavera/MS Project export',
   319→    },
   320→  })
   321→
   322→  const supervisors = await Promise.all(
   323→    SUPERVISORS.map((s) =>
   324→      db.supervisor.create({ data: { name: s.name, role: s.role, discipline: s.discipline } }),
   325→    ),
   326→  )
   327→
   328→  // Create activities and keep a DTO map for matching
   329→  const activityDtos: ScheduleActivityDTO[] = []
   330→  for (const a of ACTIVITIES) {
   331→    const [
   332→      activityId,
   333→      wbs,
   334→      discipline,
   335→      location,
   336→      activityName,
   337→      workType,
   338→      identifier,
   339→      plannedStart,
   340→      plannedFinish,
   341→      description,
   342→    ] = a
   343→    const searchText = buildSearchText({
   344→      discipline,
   345→      wbs,
   346→      location,
   347→      workType,
   348→      identifier,
   349→      activityName,
   350→    })
   351→    const row = await db.scheduleActivity.create({
   352→      data: {
   353→        projectId: project.id,
   354→        activityId,
   355→        wbs,
   356→        discipline,
   357→        activityName,
   358→        description,
   359→        location,
   360→        plannedStart,
   361→        plannedFinish,
   362→        actualStart: null,
   363→        actualFinish: null,
   364→        status: 'Not Started',
   365→        searchText,
   366→      },
   367→    })
   368→    activityDtos.push({
   369→      id: row.id,
   370→      activityId: row.activityId,
   371→      wbs: row.wbs,
   372→      discipline: row.discipline,
   373→      activityName: row.activityName,
   374→      description: row.description,
   375→      location: row.location,
   376→      plannedStart: row.plannedStart,
   377→      plannedFinish: row.plannedFinish,
   378→      actualStart: row.actualStart,
   379→      actualFinish: row.actualFinish,
   380→      status: row.status,
   381→      searchText: row.searchText,
   382→    })
   383→  }
   384→
   385→  const index = buildIndexFromActivities(activityDtos)
   386→  const activityByCode = new Map(activityDtos.map((a) => [a.activityId, a]))
   387→
   388→  // Demo field reports + resolution
   389→  for (const r of DEMO_REPORTS) {
   390→    const supervisor = supervisors[r.supervisorIdx]
   391→    const report = await db.fieldReport.create({
   392→      data: {
   393→        projectId: project.id,
   394→        supervisorId: supervisor.id,
   395→        supervisorName: supervisor.name,
   396→        discipline: supervisor.discipline,
   397→        inputType: r.inputType,
   398→        rawContent: r.raw,
   399→        reportDate: r.reportDate,
   400→      },
   401→    })
   402→
   403→    const candidates = retrieveCandidates(r.event, activityDtos, index)
   404→    const verdict = decide(candidates, r.event)
   405→
   406→    const executionEvent = await db.executionEvent.create({
   407→      data: {
   408→        reportId: report.id,
   409→        discipline: r.event.discipline ?? '',
   410→        workType: r.event.workType ?? '',
   411→        identifier: r.event.identifier ?? '',
   412→        location: r.event.location ?? '',
   413→        actualStart: r.event.actualStart,
   414→        actualFinish: r.event.actualFinish,
   415→        status: r.event.status ?? '',
   416→        quantity: r.event.quantity,
   417→        unit: r.event.unit,
   418→        evidence: r.event.evidence,
   419→      },
   420→    })
   421→
   422→    // Persist candidates (top + alternatives)
   423→    const topMatchRow = candidates[0]
   424→      ? await db.activityMatch.create({
   425→          data: {
   426→            executionEventId: executionEvent.id,
   427→            activityId: candidates[0].activityId,
   428→            scheduleActivityRowId: activityByCode.get(candidates[0].activityId)?.id ?? null,
   429→            semanticScore: candidates[0].signals.semantic,
   430→            identifierScore: candidates[0].signals.identifier,
   431→            disciplineScore: candidates[0].signals.discipline,
   432→            workTypeScore: candidates[0].signals.workType,
   433→            dateScore: candidates[0].signals.date,
   434→            finalScore: candidates[0].finalScore,
   435→            candidateMargin: verdict.candidateMargin,
   436→            rank: 1,
   437→            isTop: true,
   438→            explanation: JSON.stringify(verdict.explanation),
   439→          },
   440→        })
   441→      : null
   442→
   443→    for (const c of candidates.slice(1)) {
   444→      await db.activityMatch.create({
   445→        data: {
   446→          executionEventId: executionEvent.id,
   447→          activityId: c.activityId,
   448→          scheduleActivityRowId: activityByCode.get(c.activityId)?.id ?? null,
   449→          semanticScore: c.signals.semantic,
   450→          identifierScore: c.signals.identifier,
   451→          disciplineScore: c.signals.discipline,
   452→          workTypeScore: c.signals.workType,
   453→          dateScore: c.signals.date,
   454→          finalScore: c.finalScore,
   455→          candidateMargin: 0,
   456→          rank: c.rank,
   457→          isTop: false,
   458→          explanation: JSON.stringify(c.explanation),
   459→        },
   460→      })
   461→    }
   462→
   463→    await db.auditLog.create({
   464→      data: {
   465→        entityType: 'FieldReport',
   466→        entityId: report.id,
   467→        action: 'SUBMITTED',
   468→        actor: supervisor.name,
   469→        metadata: JSON.stringify({ inputType: r.inputType, discipline: supervisor.discipline }),
   470→      },
   471→    })
   472→    await db.auditLog.create({
   473→      data: {
   474→        entityType: 'FieldReport',
   475→        entityId: report.id,
   476→        action: 'RESOLVED',
   477→        actor: 'NEXORA-AI',
   478→        metadata: JSON.stringify({
   479→          decision: verdict.decision,
   480→          selectedActivityId: verdict.selectedActivityId,
   481→          topScore: verdict.topScore,
   482→          candidateMargin: verdict.candidateMargin,
   483→        }),
   484→      },
   485→    })
   486→
   487→    // Pre-approve a report to demonstrate the schedule update + audit trail
   488→    if (r.approved && topMatchRow && verdict.selectedActivityId) {
   489→      const targetActivity = activityByCode.get(verdict.selectedActivityId)
   490→      if (targetActivity) {
   491→        const decisionRow = await db.plannerDecision.create({
   492→          data: {
   493→            matchId: topMatchRow.id,
   494→            decision: 'APPROVED',
   495→            aiSuggestedActivityId: verdict.selectedActivityId,
   496→            selectedActivityId: verdict.selectedActivityId,
   497→            scheduleActivityRowId: targetActivity.id,
   498→            plannerName: r.plannerName ?? 'Arun Sharma',
   499→            reason: r.reason ?? null,
   500→          },
   501→        })
   502→        await db.scheduleActivity.update({
   503→          where: { id: targetActivity.id },
   504→          data: {
   505→            actualStart: r.event.actualStart,
   506→            actualFinish: r.event.actualFinish,
   507→            status:
   508→              r.event.status === 'Completed'
   509→                ? 'Completed'
   510→                : r.event.status === 'In Progress'
   511→                  ? 'In Progress'
   512→                  : r.event.status === 'Delayed'
   513→                    ? 'Delayed'
   514→                    : 'Completed',
   515→          },
   516→        })
   517→        await db.auditLog.create({
   518→          data: {
   519→            entityType: 'PlannerDecision',
   520→            entityId: decisionRow.id,
   521→            action: 'APPROVED',
   522→            actor: r.plannerName ?? 'Arun Sharma',
   523→            metadata: JSON.stringify({
   524→              activityId: targetActivity.activityId,
   525→              activityName: targetActivity.activityName,
   526→              aiSuggested: verdict.selectedActivityId,
   527→              selected: verdict.selectedActivityId,
   528→              actualStart: r.event.actualStart,
   529→              actualFinish: r.event.actualFinish,
   530→              status: r.event.status,
   531→              before: { actualStart: null, actualFinish: null, status: 'Not Started' },
   532→              after: {
   533→                actualStart: r.event.actualStart,
   534→                actualFinish: r.event.actualFinish,
   535→                status: r.event.status,
   536→              },
   537→            }),
   538→          },
   539→        })
   540→        await db.auditLog.create({
   541→          data: {
   542→            entityType: 'ScheduleActivity',
   543→            entityId: targetActivity.id,
   544→            action: 'UPDATED',
   545→            actor: r.plannerName ?? 'Arun Sharma',
   546→            metadata: JSON.stringify({
   547→              activityId: targetActivity.activityId,
   548→              actualFinish: r.event.actualFinish,
   549→              status: r.event.status,
   550→            }),
   551→          },
   552→        })
   553→      }
   554→    }
   555→  }
   556→
   557→  return {
   558→    project: project.code,
   559→    activities: activityDtos.length,
   560→    supervisors: supervisors.length,
   561→    reports: DEMO_REPORTS.length,
   562→    groundTruth: GROUND_TRUTH.length,
   563→  }
   564→}
